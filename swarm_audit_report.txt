class PolicyOrchestratorAgent:
    def run_swarm(self, cg_key, ls_key, golden_run=False):
        run_id = f"run_{int(time.time())}"
        report_dir = f"reports/live_runs/{run_id}"
        os.makedirs(report_dir, exist_ok=True)
        
        logger.info(f"[{self.__class__.__name__}] Starting swarm run: {run_id}")
        
        # 1. Data Collection Phase - Running in Parallel Streams
        # Stream A: Web Discovery (DiscoveryAgent)
        # Stream B: Legislative & Regulatory API Monitors
        
        # Stream A: Web Discovery
        logger.info(f"[{self.__class__.__name__}] Starting Stream A: Web Discovery")
        discovery_out = DiscoveryAgent().execute_task({"run_id": run_id})
        web_records = discovery_out.get("records", [])
        logger.info(f"[{self.__class__.__name__}] Stream A Complete: Discovered {len(web_records)} web records")
        
        # Stream B: Legislative & Regulatory API Monitors
        logger.info(f"[{self.__class__.__name__}] Starting Stream B: Legislative & Regulatory API Monitors")
        
        # Import and initialize API monitors
        from agents.legislative_monitor import LegislativeMonitorAgent
        from infrastructure.connectors.federal_register_connector import FederalRegisterConnector
        from infrastructure.connectors.regulations_gov_connector import RegulationsGovConnector
        from infrastructure.connectors.cfpb_connector import CFPBConnector
        
        # Collect API monitor data
        api_records = []
        
        # Legislative Monitor (Congress.gov + LegiScan)
        try:
            legislative_monitor = LegislativeMonitorAgent()
            leg_out = legislative_monitor.execute_task({"run_id": run_id, "cg_key": cg_key, "ls_key": ls_key})
            leg_records = leg_out.get("records", [])
            api_records.extend(leg_records)
            logger.info(f"[{self.__class__.__name__}] Legislative Monitor: Collected {len(leg_records)} records")
        except Exception as e:
            logger.error(f"[{self.__class__.__name__}] Legislative Monitor failed: {e}")
        
        # Federal Register Monitor
        try:
            fr_config = {
                'api_base_url': 'https://www.federalregister.gov/api/v1/articles.json',
                'timeout': 30,
                'rate_limit_delay': 1.0
            }
            fr_connector = FederalRegisterConnector(fr_config)
            if fr_connector.connect() and fr_connector.authenticate():
                fr_data = fr_connector.incremental_sync()
                # Convert raw data to records format
                for item in fr_data if isinstance(fr_data, list) else []:
                    api_records.append({
                        "title": item.get('title', 'Federal Register Article'),
                        "summary": item.get('abstract', '')[:500] + "..." if len(item.get('abstract', '')) > 500 else item.get('abstract', ''),
                        "source": "Federal Register",
                        "url": item.get('html_url', ''),
                        "date": item.get('publication_date', '')
                    })
                fr_connector.shutdown()
                logger.info(f"[{self.__class__.__name__}] Federal Register Monitor: Collected {len(fr_data) if isinstance(fr_data, list) else 0} records")
        except Exception as e:
            logger.error(f"[{self.__class__.__name__}] Federal Register Monitor failed: {e}")
        
        # Regulations.gov Monitor
        try:
            regs_config = {
                'api_key': os.getenv('REGULATIONS_GOV_API_KEY', 'DEMO_KEY'),
                'timeout': 30,
                'rate_limit_delay': 1.0
            }
            regs_connector = RegulationsGovConnector(regs_config)
            if regs_connector.connect() and regs_connector.authenticate():
                regs_data = regs_connector.incremental_sync()
                # Convert raw data to records format
                for item in regs_data if isinstance(regs_data, list) else []:
                    attributes = item.get('attributes', {}) if isinstance(item, dict) else {}
                    api_records.append({
                        "title": attributes.get('title', 'Regulations.gov Document'),
                        "summary": f"Type: {attributes.get('documentType', 'Document')}",
                        "source": "Regulations.gov",
                        "url": attributes.get('fileFormats', [{}])[0].get('fileUrl', f"https://www.regulations.gov/document/{attributes.get('objectId', '')}") if attributes.get('fileFormats') and len(attributes.get('fileFormats', [])) > 0 else f"https://www.regulations.gov/document/{attributes.get('objectId', '')}",
                        "date": attributes.get('postedDate', '')
                    })
                regs_connector.shutdown()
                logger.info(f"[{self.__class__.__name__}] Regulations.gov Monitor: Collected {len(regs_data) if isinstance(regs_data, list) else 0} records")
        except Exception as e:
            logger.error(f"[{self.__class__.__name__}] Regulations.gov Monitor failed: {e}")
        
        # CFPB Monitor
        try:
            cfpb_config = {
                'api_base_url': 'https://data.consumerfinance.gov/resource/jhzv-w97w.json',
                'timeout': 30,
                'rate_limit_delay': 1.0
            }
            cfpb_connector = CFPBConnector(cfpb_config)
            if cfpb_connector.connect():
                cfpb_data = cfpb_connector.incremental_sync()
                cfpb_records = cfpb_data.get('new_data', []) if isinstance(cfpb_data, dict) else []
                # Convert raw data to records format
                for item in cfpb_records:
                    api_records.append({
                        "title": f"{item.get('product', '')} - {item.get('issue', '')}".strip(' -'),
                        "summary": (item.get('consumer_complaint_narrative', '')[:500] + "...") if len(item.get('consumer_complaint_narrative', '')) > 500 else item.get('consumer_complaint_narrative', ''),
                        "source": "CFPB",
                        "url": f"https://www.consumerfinance.gov/data-research/consumer-complaints/search/?complaint_id={item.get('complaint_id', '')}" if item.get('complaint_id') else "https://www.consumerfinance.gov/data-research/consumer-complaints/",
                        "date": item.get('date_received', '')
                    })
                cfpb_connector.shutdown()
                logger.info(f"[{self.__class__.__name__}] CFPB Monitor: Collected {len(cfbp_records)} records")
        except Exception as e:
            logger.error(f"[{self.__class__.__name__}] CFPB Monitor failed: {e}")
        
        logger.info(f"[{self.__class__.__name__}] Stream B Complete: Collected {len(api_records)} API monitor records")
        
        # Merge all outputs into a single list of records
        all_records = web_records + api_records
        logger.info(f"[{self.__class__.__name__}] Total combined records: {len(all_records)} ({len(web_records)} web + {len(api_records)} API)")
        
        # Apply golden run limits: exactly 1 live item from discovery
        if golden_run:
            # Take only the first record for golden run
            all_records = all_records[:1]
            logger.info(f"[{self.__class__.__name__}] Golden run mode: limited to {len(all_records)} record")
        
        # 1.5. Context Expansion (Inserted after data collection, before analysis)
        context_out = ContextExpansionAgent().execute_task({"records": all_records, "run_id": run_id})
        expanded_records = context_out.get("records", [])
        logger.info(f"[{self.__class__.__name__}] Expanded context for {len(expanded_records)} records")
        
        # 2. Analyze
        impact_out = ImpactAnalyzerAgent().execute_task({"records": expanded_records, "run_id": run_id})
        findings = impact_out.get("records", [])
        logger.info(f"[{self.__class__.__name__}] Analyzed {len(findings)} records")
        
        # 2.5. Simulated Human Feedback Loop (for golden dataset generation)
        # Select one record rejected by ImpactAnalyzer (low impact score) for simulated human review
        low_impact_records = [r for r in findings if r.get("impact_score") == "low"]
        if low_impact_records:
            # Select the first low-impact record for simulation
            record_for_feedback = low_impact_records[0]
            # Simulate human override: mark as high impact with corrected reasoning
            human_decision = "monitor"
            corrected_reasoning = "Upon manual review, this record contains significant BNPL relevance despite initial low-score assessment. Contains relevant credit finance implications."
            
            # Instantiate and use HumanFeedbackAgent
            feedback_agent = HumanFeedbackAgent()
            feedback_result = feedback_agent.execute_task({
                "record": record_for_feedback,
                "human_decision": human_decision,
                "corrected_reasoning": corrected_reasoning
            })
            logger.info(f"[{self.__class__.__name__}] Human feedback logged: {feedback_result}")
        else:
            logger.info(f"[{self.__class__.__name__}] No low-impact records found for simulated human feedback")
        
        # 3. Context Compression (Map)
        compression_out = ContextCompressorAgent().execute_task({"findings": findings, "run_id": run_id})
        compressed_findings = compression_out.get("findings", [])
        logger.info(f"[{self.__class__.__name__}] Compressed {len(compressed_findings)} findings")
        
        # 4. Entity Resolution (Extract entities from compressed findings)
        entity_out = EntityResolutionAgent().execute_task({"findings": compressed_findings, "run_id": run_id})
        entity_enriched_findings = entity_out.get("findings", [])
        logger.info(f"[{self.__class__.__name__}] Extracted entities for {len(entity_enriched_findings)} findings")
        
        # 5. Risk Scoring (Score the entity-enriched findings)
        risk_out = RiskScoringAgent().execute_task({"findings": entity_enriched_findings, "run_id": run_id})
        risk_scored_findings = risk_out.get("findings", [])
        logger.info(f"[{self.__class__.__name__}] Scored risk for {len(risk_scored_findings)} findings")
        
        # 6. Validate
        val_out = ValidationAgent().execute_task({"findings": risk_scored_findings, "run_id": run_id})
        validated_findings = val_out.get("findings", [])
        logger.info(f"[{self.__class__.__name__}] Validated {len(validated_findings)} findings")
        
        # 6.5. Human Review Queue for gray area (70-79 risk score)
        gray_area_findings = [f for f in validated_findings if 70 <= f.get('risk_score', 0) <= 79]
        if gray_area_findings:
            review_queue_path = os.path.join(report_dir, "review_queue.json")
            with open(review_queue_path, 'w') as f:
                json.dump(gray_area_findings, f, indent=2)
            logger.info(f"[{self.__class__.__name__}] Created review queue with {len(gray_area_findings)} items at {review_queue_path}")

        # 7. Sync to Graph Database
        try:
            graph_connector = GraphConnector()
            graph_connector.sync_findings(validated_findings, run_id)
            graph_connector.close()
            logger.info(f"[{self.__class__.__name__}] Synced {len(validated_findings)} findings to Neo4j")
        except Exception as e:
            logger.error(f"[{self.__class__.__name__}] Failed to sync to Neo4j: {e}")
            # Continue with the swarm even if graph sync fails
        
        # 8. Synthesize (using compressed findings)
        synth_out = ResearchSynthesisAgent().execute_task({"findings": compressed_findings, "run_id": run_id})
        logger.info(f"[{self.__class__.__name__}] Synthesis complete")
        
        # 9. Report (The final step) - using validated findings with entities and scores
        ReportingAgent().execute_task({
            "report_dir": report_dir,
            "findings": validated_findings,
            "synthesis": synth_out
        })
        logger.info(f"[{self.__class__.__name__}] Report generated in {report_dir}")
        
        return {"report_dir": report_dir}

    def update_source_health(self, source_name: str, success: bool, error: str = None):
        if source_name in self._catalog:
            # For senate_banking_committee, treat 503 as temporary unavailability
            source_config = self._catalog[source_name]
            if source_name == "senate_banking_committee" and error and "503" in str(error):
                # Log as temporary issue but don't mark as failed
                source_config["last_error"] = "Source temporarily unavailable (503)"
                # Don't update last_success to avoid marking as failed
            else:
                self._catalog[source_name]["last_success"] = datetime.datetime.now().isoformat() if success else self._catalog[source_name]["last_success"]
                self._catalog[source_name]["last_error"] = error

    def list_enabled_sources(self) -> list:
        return [name for name, meta in self._catalog.items() if meta["enabled"]]

    def get_enabled_sources_by_type(self, source_type: str) -> list:
        return [name for name, meta in self._catalog.items() if meta["enabled"] and meta["source_type"] == source_type]

    def get_fallback_data(self, source_name: str):
        """Get fallback data for a source when it's unavailable"""
        source_config = self._catalog.get(source_name, {})
        if source_config.get("fallback_enabled") and "fallback_data" in source_config:
            return source_config["fallback_data"]
        return []
