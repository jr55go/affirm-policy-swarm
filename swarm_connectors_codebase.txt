145	    def fetch(self, query: Dict[str, Any]) -> List[Dict[str, Any]]:
146	        """
147	        Retrieve data from the Regulations.gov API based on query parameters.
148	        
149	        Args:
150	            query: Dictionary containing query parameters
151	                  Expected keys: 'size', 'filter', 'sort', etc.
152	                  
153	        Returns:
154	            List of dictionaries containing the retrieved data
155	        """
156	        if not self.is_connected or not self._session:
157	            self.connect()
158	            self.authenticate()
159	        
160	        try:
161	            self.logger.info(f"Fetching data from Regulations.gov API with query: {query}")
162	            
163	            # Prepare request parameters for Regulations.gov API
164	            params = {}
165	            
166	            # Handle size parameter -> page[size]
167	            if 'size' in query:
168	                # Enforce minimum page[size] of 5 for Regulations.gov API
169	                requested_size = int(query['size'])
170	                params['page[size]'] = max(5, requested_size)
171	            else:
172	                params['page[size]'] = 100  # default batch size (already >= 5)
173	            
174	            # Handle filter parameters (Regulations.gov uses filter[field]=value format)
175	            if 'filter' in query and isinstance(query['filter'], dict):
176	                for key, value in query['filter'].items():
177	                    params[f'filter[{key}]'] = value
178	            
179	            # Handle search term parameters (q, term, query) -> map to filter[searchTerm]
180	            search_keys = ['q', 'term', 'query']
181	            for key in search_keys:
182	                if key in query and query[key]:
183	                    params['filter[searchTerm]'] = query[key]
184	                    break  # Only use the first one found
185	            
186	            # Handle sort parameter - always ensure sort is present
187	            if 'sort' in query:
188	                params['sort'] = query['sort']
189	            else:
190	                # Default sort as required by the API
191	                params['sort'] = '-lastModifiedDate'
192	            
193	            # Crucial: Always ensure the final parameters contain "sort": "-lastModifiedDate"
194	            # (already handled above by setting default if not present)
195	            
196	            # Crucial: If the final parameters contain no filter[...] keys at all, 
197	            # attach a default filter to satisfy API constraints
198	            has_filter = any(key.startswith('filter[') for key in params.keys())
199	            if not has_filter:
200	                # Add default filter for document type: Rule
201	                params['filter[documentType]'] = 'Rule'
202	            
203	            # Make the request
204	            response = self._session.get(
205	                self.api_base_url,
206	                params=params,
207	                timeout=self.timeout
208	            )
209	            
210	            # Raise exception for bad status codes
211	            response.raise_for_status()
212	            
213	            # Parse JSON response
214	            response_data = response.json()
215	            
216	            # Regulations.gov API returns a dictionary with 'data' array and metadata
217	            raw_data = response_data.get('data', [])
218	            
219	            # Ensure we got a list
220	            if not isinstance(raw_data, list):
221	                self.logger.warning(f"Expected list from Regulations.gov API, got {type(raw_data)}")
222	                raw_data = []
223	            
224	            self.logger.info(f"Retrieved {len(raw_data)} records from Regulations.gov API")
225	            return raw_data
226	            
227	        except RequestException as e:
228	            self.logger.error(f"Request to Regulations.gov API failed: {str(e)}")
229	            return []
230	        except Exception as e:
231	            self.logger.error(f"Unexpected error fetching data from Regulations.gov API: {str(e)}")
232	            return []

235	        """
236	        Normalize retrieved Regulations.gov data to the NormalizedPolicyEvent schema.
237	        
238	        Args:
239	            raw_data: Dictionary containing raw Regulations.gov document data
240	            
241	        Returns:
242	            NormalizedPolicyEvent object containing normalized policy event data
243	        """
244	        try:
245	            # Regulations.gov API v4 document structure:
246	            # {
247	            #   "data": {
248	            #     "type": "documents",
249	            #     "id": "document-id",
250	            #     "attributes": {
251	            #       "title": "Document Title",
252	            #       "documentType": "Rule",
253	            #       "postedDate": "2023-01-01T00:00:00Z",
254	            #       "objectId": "object-id",
255	            #       "fileFormats": [
256	            #         {
257	            #           "fileUrl": "https://example.com/document.pdf",
258	            #           "fileFormat": "PDF",
259	            #           "fileSize": 12345
260	            #         }
261	            #       ],
262	            #       "agencyId": ["agency1", "agency2"]
263	            #     }
264	            #   }
265	            # }
266	            
267	            # Extract attributes from the data structure
268	            attributes = raw_data.get('attributes', {}) if isinstance(raw_data, dict) else {}
269	            
270	            # Extract fields according to the mapping requirements
271	            title = attributes.get('title', '')
272	            document_type = attributes.get('documentType', '')  # This maps to event_type
273	            posted_date = attributes.get('postedDate', '')  # This maps to event_date
274	            
275	            # Extract URL: prefer fileFormats[0].fileUrl, else construct from objectId
276	            url = ''
277	            file_formats = attributes.get('fileFormats', [])
278	            if isinstance(file_formats, list) and len(file_formats) > 0:
279	                first_format = file_formats[0]
280	                if isinstance(first_format, dict) and 'fileUrl' in first_format:
281	                    url = first_format['fileUrl']
282	            
283	            # If no file URL found, construct web URL using objectId
284	            if not url:
285	                object_id = attributes.get('objectId', '')
286	                if object_id:
287	                    url = f"https://www.regulations.gov/document/{object_id}"
288	            
289	            # Extract agency IDs
290	            agency_ids = attributes.get('agencyId', [])
291	            if isinstance(agency_ids, str):
292	                agency_ids = [agency_ids]
293	            elif not isinstance(agency_ids, list):
294	                agency_ids = []
295	            
296	            # Construct summary from available fields
297	            summary_parts = []
298	            if title:
299	                summary_parts.append(title)
300	            if document_type:
301	                summary_parts.append(f"Type: {document_type}")
302	            summary = " - ".join(summary_parts) if summary_parts else "Regulations.gov document"
303	            
304	            # Parse the posted date
305	            from datetime import datetime
306	            try:
307	                # Handle ISO format with timezone
308	                if posted_date:
309	                    # Remove timezone info for simpler parsing if needed
310	                    date_str = posted_date.replace('Z', '+00:00')
311	                    event_date = datetime.fromisoformat(date_str)
312	                else:
313	                    event_date = datetime.now()
314	            except ValueError:
315	                # Fallback to current time if parsing fails
316	                event_date = datetime.now()
317	            
318	            # Create normalized event using the schema
319	            normalized_event = NormalizedPolicyEvent(
320	                source_name="Regulations.gov",
321	                source_type="Regulator",
322	                jurisdiction="Federal",
323	                title=title,
324	                summary=summary,
325	                event_type=document_type,
326	                event_date=event_date,
327	                url=url,
328	                organizations=[],  # Regulations.gov documents don't typically have separate organizations field
329	                agencies=agency_ids,  # Map agencyId to agencies list
330	                committees=[],  # Regulations.gov doesn't involve congressional committees
331	                legislators=[],  # Regulations.gov doesn't involve specific legislators
332	                keywords=[],  # Could extract from title/description but not required for now
333	                confidence=0.9,  # High confidence since we're mapping directly from official API
334	                raw_payload=raw_data  # Store the original document data
335	            )
336	
337	            return normalized_event
338	                
339	        except Exception as e:
340	            self.logger.error(f"Error normalizing Regulations.gov document: {str(e)}")
341	            # Return a minimal valid normalized event in case of error
342	            from datetime import datetime
343	            return NormalizedPolicyEvent(
344	                source_name="Regulations.gov",
345	                source_type="Regulator",
346	                jurisdiction="Federal",
347	                title="Error processing Regulations.gov document",
348	                summary=f"Failed to normalize Regulations.gov document: {str(e)}",
349	                event_type="error",
350	                event_date=datetime.now(),
351	                url="",
352	                organizations=[],
353	                agencies=[],
354	                committees=[],
355	                legislators=[],
356	                keywords=[],
357	                confidence=0.0,
358	                raw_payload=raw_data
359	            )

410	    def retry(self, operation: callable, max_retries: int = 3, backoff_factor: float = 1.0) -> Any:
411	        """
412	        Retry a failed operation with exponential backoff.
413	        
414	        Args:
415	            operation: Callable operation to retry
416	            max_retries: Maximum number of retry attempts
417	            backoff_factor: Backoff factor for exponential delay
418	            
419	        Returns:
420	            Result of the operation if successful
421	            
422	        Raises:
423	            Exception: If all retry attempts fail
424	        """
425	        last_exception = None
426	        
427	        for attempt in range(max_retries):
428	            try:
429	                self.logger.debug(f"Attempt {attempt + 1} of {max_retries} for operation")
430	                result = operation()
431	                if attempt > 0:
432	                    self.logger.info(f"Operation succeeded on attempt {attempt + 1}")
433	                return result
434	            except Exception as e:
435	                last_exception = e
436	                self.logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
437	                if attempt < max_retries - 1:  # Don't sleep on the last attempt
438	                    sleep_time = backoff_factor * (2 ** attempt)
439	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
440	                    time.sleep(sleep_time)
441	        
442	        self.logger.error(f"All {max_retries} attempts failed. Last error: {str(last_exception)}")
443	        raise last_exception

1	import heapq
2	import time
3	from typing import Dict, List, Tuple, Optional
4	from dataclasses import dataclass, field
5	from enum import IntEnum
6	
7	
8	class Tier(IntEnum):
9	    """Priority tiers for connector scheduling."""
10	    TIER_1 = 1  # Highest priority - most frequent
11	    TIER_2 = 2
12	    TIER_3 = 3
13	    TIER_4 = 4
14	    TIER_5 = 5
15	    TIER_6 = 6  # Lowest priority - least frequent
16	
17	
18	@dataclass
19	class ScheduledTask:
20	    """Represents a scheduled connector run."""
21	    source_name: str
22	    tier: Tier
23	    interval_seconds: int
24	    next_run: float = field(default_factory=time.time)
25	    last_run: Optional[float] = None
26	    run_count: int = 0
27	    
28	    def __lt__(self, other):
29	        """For priority queue ordering - lower next_run time has higher priority."""
30	        return self.next_run < other.next_run
31	
32	
33	class Scheduler:
34	    """Manages scheduling of connector runs based on tier priorities."""
35	    
36	    # Base intervals in seconds for each tier (can be overridden in config)
37	    BASE_INTERVALS = {
38	        Tier.TIER_1: 3600,      # 1 hour
39	        Tier.TIER_2: 7200,      # 2 hours
40	        Tier.TIER_3: 21600,     # 6 hours
41	        Tier.TIER_4: 43200,     # 12 hours
42	        Tier.TIER_5: 86400,     # 1 day
43	        Tier.TIER_6: 604800,    # 1 week
44	    }
45	    
46	    def __init__(self):
47	        self._scheduled_tasks: Dict[str, ScheduledTask] = {}
48	        self._priority_queue: List[ScheduledTask] = []
49	    
50	    def register_source(self, source_name: str, tier: Tier, custom_interval: Optional[int] = None) -> None:
51	        """Register a source for scheduling.
52	        
53	        Args:
54	            source_name: Name of the data source
55	            tier: Priority tier (1-6)
56	            custom_interval: Optional custom interval in seconds (overrides tier default)
57	        """
58	        interval = custom_interval if custom_interval is not None else self.BASE_INTERVALS[tier]
59	        now = time.time()
60	        
61	        task = ScheduledTask(
62	            source_name=source_name,
63	            tier=tier,
64	            interval_seconds=interval,
65	            next_run=now,
66	            last_run=None,
67	            run_count=0
68	        )
69	        
70	        self._scheduled_tasks[source_name] = task
71	        heapq.heappush(self._priority_queue, task)
72	    
73	    def get_next_due(self) -> Optional[Tuple[str, float]]:
74	        """Get the next source that is due to run.
75	        
76	        Returns:
77	            Tuple of (source_name, seconds_until_due) or None if no tasks
78	        """
79	        if not self._priority_queue:
80	            return None
81	            
82	        # Peek at the next task without popping
83	        now = time.time()
84	        next_task = self._priority_queue[0]
85	        
86	        if next_task.next_run <= now:
87	            return (next_task.source_name, 0.0)
88	        else:
89	            return (next_task.source_name, next_task.next_run - now)
90	    
91	    def mark_run(self, source_name: str) -> None:
92	        """Mark a source as having just run, rescheduling it.
93	        
94	        Args:
95	            source_name: Name of the source that just ran
96	        """
97	        if source_name not in self._scheduled_tasks:
98	            return
99	            
100	        task = self._scheduled_tasks[source_name]
101	        now = time.time()
102	        
103	        # Update the task
104	        task.last_run = now
105	        task.run_count += 1
106	        task.next_run = now + task.interval_seconds
107	        
108	        # Re-heapify the priority queue since the priority changed
109	        heapq.heapify(self._priority_queue)
110	    
111	    def get_status(self) -> Dict[str, dict]:
112	        """Get the current status of all scheduled sources.
113	        
114	        Returns:
115	            Dictionary mapping source names to their status information
116	        """
117	        now = time.time()
118	        status = {}
119	        
120	        for source_name, task in self._scheduled_tasks.items():
121	            status[source_name] = {
122	                "tier": task.tier.name,
123	                "interval_seconds": task.interval_seconds,
124	                "last_run": task.last_run,
125	                "next_run": task.next_run,
126	                "seconds_until_next": max(0, task.next_run - now),
127	                "run_count": task.run_count
128	            }
129	            
130	        return status

109	    def fetch(self, query: Dict[str, Any]) -> List[Dict[str, Any]]:
110	        """
111	        Retrieve data from the Federal Register API based on query parameters.
112	
113	        Args:
114	            query: Dictionary containing query parameters
115	                  Expected keys: 'size', 'start_date', 'end_date', 'type', 'agency_names', etc.
116	
117	        Returns:
118	            List of dictionaries containing the retrieved data
119	        """
120	        if not self.is_connected or not self._session:
121	            self.connect()
122	            self.authenticate()
123	
124	        try:
125	            self.logger.info(f"Fetching data from Federal Register API with query: {query}")
126	
127	            # Prepare request parameters for Federal Register API
128	            params = {}
129	
130	            # Handle size parameter -> per_page
131	            if 'size' in query:
132	                params['per_page'] = query['size']
133	            else:
134	                params['per_page'] = 100  # default batch size
135	
136	            # Date range filtering (Federal Register uses publication_date)
137	            if 'start_date' in query:
138	                params['publication_date[gte]'] = query['start_date']
139	            if 'end_date' in query:
140	                params['publication_date[lte]'] = query['end_date']
141	
142	            # Type filtering (e.g., Rule, Proposed Rule, Notice)
143	            if 'type' in query:
144	                params['type'] = query['type']
145	
146	            # Agency filtering
147	            if 'agency_names' in query:
148	                # Federal Register API expects agency names as a comma-separated string for the 'agencies[]' parameter
149	                # However, the API uses a different format: we can use 'agencies[]' multiple times or a comma-separated string in one parameter
150	                # Let's use the format: agencies[]=agency1&agencies[]=agency2
151	                # But requests will handle a list of values for the same key if we pass a list
152	                # We'll pass as a list for the key 'agencies[]'
153	                agency_names = query['agency_names']
154	                if isinstance(agency_names, list):
155	                    params['agencies[]'] = agency_names
156	                else:
157	                    # If it's a string, split by comma if needed, but assume it's a single agency or comma-separated string
158	                    # We'll split by comma and make a list
159	                    agency_list = [name.strip() for name in str(agency_names).split(',') if name.strip()]
160	                    if agency_list:
161	                        params['agencies[]'] = agency_list
162	
163	            # Make the request
164	            response = self._session.get(
165	                self.api_base_url,
166	                params=params,
167	                timeout=self.timeout
168	            )
169	
170	            # Raise exception for bad status codes
171	            response.raise_for_status()
172	
173	            # Parse JSON response
174	            response_data = response.json()
175	
176	            # Federal Register API returns a dictionary with 'results' array and metadata
177	            raw_data = response_data.get('results', [])
178	
179	            # Ensure we got a list
180	            if not isinstance(raw_data, list):
181	                self.logger.warning(f"Expected list from Federal Register API, got {type(raw_data)}")
182	                raw_data = []
183	
184	            self.logger.info(f"Retrieved {len(raw_data)} records from Federal Register API")
185	            return raw_data
186	
187	        except RequestException as e:
188	            self.logger.error(f"Request to Federal Register API failed: {str(e)}")
189	            return []
190	        except Exception as e:
191	            self.logger.error(f"Unexpected error fetching data from Federal Register API: {str(e)}")
192	            return []

322	        """
323	        Retry a failed operation with exponential backoff.
324	
325	        Args:
326	            operation: Callable operation to retry
327	            max_retries: Maximum number of retry attempts
328	            backoff_factor: Backoff factor for exponential delay
329	
330	        Returns:
331	            Result of the operation if successful
332	
333	        Raises:
334	            Exception: If all retry attempts fail
335	        """
336	        last_exception = None
337	
338	        for attempt in range(max_retries):
339	            try:
340	                self.logger.debug(f"Attempt {attempt + 1} of {max_retries} for operation")
341	                result = operation()
342	                if attempt > 0:
343	                    self.logger.info(f"Operation succeeded on attempt {attempt + 1}")
344	                return result
345	            except Exception as e:
346	                last_exception = e
347	                self.logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
348	                if attempt < max_retries - 1:  # Don't sleep on the last attempt
349	                    sleep_time = backoff_factor * (2 ** attempt)
350	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
351	                    time.sleep(sleep_time)
352	
353	        self.logger.error(f"All {max_retries} attempts failed. Last error: {str(last_exception)}")
354	        raise last_exception

324	        """
325	        Retry a failed operation with exponential backoff.
326	
327	        Args:
328	            operation: Callable operation to retry
329	            max_retries: Maximum number of retry attempts
330	            backoff_factor: Backoff factor for exponential delay
331	
332	        Returns:
333	            Result of the operation if successful
334	
335	        Raises:
336	            Exception: If all retry attempts fail
337	        """
338	        last_exception = None
339	
340	        for attempt in range(max_retries):
341	            try:
342	                self.logger.debug(f"Attempt {attempt + 1} of {max_retries} for operation")
343	                result = operation()
344	                if attempt > 0:
345	                    self.logger.info(f"Operation succeeded on attempt {attempt + 1}")
346	                return result
347	            except Exception as e:
348	                last_exception = e
349	                self.logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
350	                if attempt < max_retries - 1:  # Don't sleep on the last attempt
351	                    sleep_time = backoff_factor * (2 ** attempt)
352	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
353	                    time.sleep(sleep_time)
354	
355	        self.logger.error(f"All {max_retries} attempts failed. Last error: {str(last_exception)}")
356	        raise last_exception

397	        """
398	        Retry a failed operation with exponential backoff.
399	
400	        Args:
401	            operation: Callable operation to retry
402	            max_retries: Maximum number of retry attempts
403	            backoff_factor: Backoff factor for exponential delay
404	
405	        Returns:
406	            Result of the operation if successful
407	
408	        Raises:
409	            Exception: If all retry attempts fail
410	        """
411	        last_exception = None
412	
413	        for attempt in range(max_retries):
414	            try:
415	                self.logger.debug(f"Attempt {attempt + 1} of {max_retries} for operation")
416	                result = operation()
417	                if attempt > 0:
418	                    self.logger.info(f"Operation succeeded on attempt {attempt + 1}")
419	                return result
420	            except Exception as e:
421	                last_exception = e
422	                self.logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
423	                if attempt < max_retries - 1:  # Don't sleep on the last attempt
424	                    sleep_time = backoff_factor * (2 ** attempt)
425	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
426	                    time.sleep(sleep_time)
427	
428	        self.logger.error(f"All {max_retries} attempts failed. Last error: {str(last_exception)}")
429	        raise last_exception

143	            params = {}

146	            if 'size' in query:

152	            params['$order'] = 'date_received DESC'

161	                where_conditions.append(f"date_received <= '{query['end_date']}'")
162	            
163	            # Product filtering
164	            if 'product' in query:
165	                # Escape single quotes in product string by doubling them
166	                product_safe = query['product'].replace("'", "''")
167	                where_conditions.append(f"product = '{product_safe}'")
168	                
169	            # Issue filtering
170	            if 'issue' in query:
171	                # Escape single quotes in issue string
172	                issue_safe = query['issue'].replace("'", "''")
173	                where_conditions.append(f"issue = '{issue_safe}'")
174	            
175	            # If we have any WHERE conditions, combine them
176	            if where_conditions:
177	                params['$where'] = " AND ".join(where_conditions)
178	            
179	            # Make the request
180	            response = self._session.get(
181	                self.api_base_url,
182	                params=params,
183	                timeout=self.timeout,
184	                verify=False  # Bypass SSL verification

190	                time.sleep(self.rate_limit_delay * 2)  # Back off longer on rate limit

197	            raw_data = response.json()

199	            # Ensure we got a list
200	            if not isinstance(raw_data, list):
201	                self.logger.warning(f"Expected list from Socrata API, got {type(raw_data)}")
202	                raw_data = []

205	            return raw_data

460	                    # Retry server errors

462	                else:
463	                    # Retry on connection errors, timeouts, etc.
464	                    return attempt < 3
465	            else:
466	                # Retry on connection errors
467	                return attempt < 3
468	        else:

403	        """

405	
406	        Args:
407	            operation: Callable operation to retry
408	            max_retries: Maximum number of retry attempts
409	            backoff_factor: Backoff factor for exponential delay
410	
411	        Returns:
412	            Result of the operation if successful
413	
414	        Raises:
415	            Exception: If all retry attempts fail
416	        """
417	        last_exception = None
418	
419	        for attempt in range(max_retries):
420	            try:
421	                self.logger.debug(f"Attempt {attempt + 1} of {max_retries} for operation")
422	                result = operation()
423	                if attempt > 0:
424	                    self.logger.info(f"Operation succeeded on attempt {attempt + 1}")
425	                return result
426	            except Exception as e:
427	                last_exception = e
428	                self.logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
429	                if attempt < max_retries - 1:  # Don't sleep on the last attempt
430	                    sleep_time = backoff_factor * (2 ** attempt)
431	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
432	                    time.sleep(sleep_time)
433	
434	        self.logger.error(f"All {max_retries} attempts failed. Last error: {str(last_exception)}")
435	        raise last_exception

436	                    self.logger.debug(f"Sleeping {sleep_time} seconds before retry")
437	                    time.sleep(sleep_time)

505	    def retry(self, exception: Exception, attempt: int) -> bool:
506	        """
507	        Determine whether to retry an operation after an exception.
508	        
509	        Args:
510	            exception: The exception that occurred
511	            attempt: Current attempt number (starting from 1)
512	            
513	        Returns:
514	            True if the operation should be retried, False otherwise
515	        """
516	        # Don't retry client errors (4xx) except 429 (rate limit)
517	        if isinstance(exception, RequestException):
518	            if hasattr(exception, 'response') and exception.response is not None:
519	                status_code = exception.response.status_code
520	                if 400 <= status_code < 500:
521	                    # Don't retry client errors except rate limiting
522	                    return status_code == 429 and attempt < 3
523	                elif 500 <= status_code < 600:
524	                    # Retry server errors
525	                    return attempt < 3
526	                else:
527	                    # Retry on connection errors, timeouts, etc.
528	                    return attempt < 3
529	            else:
530	                # Retry on connection errors
531	                return attempt < 3
532	        else:
533	            # Retry on other exceptions up to 3 times
534	            return attempt < 3

396	    def retry(self, exception: Exception, attempt: int) -> bool:
397	        """
398	        Determine whether to retry an operation after an exception.
399	        
400	        Args:
401	        exception: The exception that occurred
402	        attempt: Current attempt number (starting from 1)
403	            
404	        Returns:
405	        True if the operation should be retried, False otherwise
406	        """
407	        # Don't retry client errors (4xx) except 429 (rate limit)
408	        if isinstance(exception, RequestException):
409	            if hasattr(exception, 'response') and exception.response is not None:
410	                status_code = exception.response.status_code
411	                if 400 <= status_code < 500:
412	                    # Don't retry client errors except rate limiting
413	                    return status_code == 429 and attempt < 3
414	                elif 500 <= status_code < 600:
415	                    # Retry server errors
416	                    return attempt < 3
417	                else:
418	                    # Retry on connection errors, timeouts, etc.
419	                    return attempt < 3
420	            else:
421	                # Retry on connection errors
422	                return attempt < 3
423	        else:
424	            # Retry on other exceptions up to 3 times
425	            return attempt < 3
