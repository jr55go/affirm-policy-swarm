     5	class RiskScoringAgent:
     6	    def __init__(self, agent_id="RiskScoringAgent"):
     7	        self.agent_id = agent_id
     8	        self.ollama_url = "http://localhost:11434/api/generate"
     9	        self.model = "qwen2.5-coder:32b"
    10	
    11	    def _call_ollama(self, prompt):
    12	        payload = {
    13	            "model": self.model,
    14	            "prompt": prompt,
    15	            "stream": False
    16	        }
    17	        try:
    18	            response = requests.post(self.ollama_url, json=payload, timeout=60)
    19	            response.raise_for_status()
    20	            return response.json().get("response", "")
    21	        except Exception as e:
    22	            print(f"[{self.agent_id}] Error calling Ollama: {e}")
    23	            return None
    24	
    25	    def _score_risk_with_llm(self, finding, text, entities):
    26	        """
    27	        Use the LLM to score the risk and update the finding dictionary with:
    28	        - policy_risk_score (int)
    29	        - alert_required (bool)
    30	        - risk_reasoning (str)
    31	        
    32	        Returns the updated finding dictionary.
    33	        """
    34	        if not text or not text.strip():
    35	            finding["policy_risk_score"] = 0
    36	            finding["alert_required"] = False
    37	            finding["risk_reasoning"] = "No text to analyze"
    38	            return finding
    39	
    40	        # Format entities for the prompt
    41	        orgs = ", ".join(entities.get("organizations", [])) if entities.get("organizations") else "none"
    42	        policymakers = ", ".join(entities.get("policymakers", [])) if entities.get("policymakers") else "none"
    43	        themes = ", ".join(entities.get("themes", [])) if entities.get("themes") else "none"
    44	
    45	        prompt = f"""
    46	You are a regulatory risk analyst. Evaluate the following policy finding for risk.
    47	
    48	Finding Summary:
    49	{text}
    50	
    51	Entities Mentioned:
    52	- Organizations: {orgs}
    53	- Policymakers: {policymakers}
    54	- Themes: {themes}
    55	
    56	Based on the summary and entities, assess the risk level considering:
    57	1. Entity influence (e.g., CFPB, SEC, major legislators)
    58	2. Policy momentum (e.g., active legislation, recent votes, enforcement actions)
    59	3. Market relevance (e.g., impact on consumer finance, BNPL, lending)
    60	
    61	Return a strict JSON object with exactly these three keys:
    62	- "policy_risk_score": an integer from 0 to 100 representing the risk score
    63	- "alert_required": a boolean indicating if an alert should be triggered (true if score >= 80)
    64	- "risk_reasoning": a string explaining the reasoning for the score
    65	
    66	Example:
    67	{{
    68	  "policy_risk_score": 85,
    69	  "alert_required": true,
    70	  "risk_reasoning": "Direct CFPB enforcement action mentioned with potential market-wide impact on BNPL providers."
    71	}}
    72	
    73	Only return the JSON object.
    74	""".strip()
    75	
    76	        llm_response = self._call_ollama(prompt)
    77	        if llm_response is None:
    78	            print(f"[{self.agent_id}] LLM call failed for risk scoring. Using fallback.")
    79	            return self._fallback_scoring(finding, text, entities)
    80	
    81	        # Try to parse the JSON response
    82	        try:
    83	            # Find the JSON object in the response (in case there's extra text)
    84	            start_idx = llm_response.find('{')
    85	            end_idx = llm_response.rfind('}')
    86	            if start_idx != -1 and end_idx != -1:
    87	                json_str = llm_response[start_idx:end_idx+1]
    88	                result = json_module.loads(json_str)
    89	                # Validate and set defaults if needed
    90	                score = result.get("policy_risk_score", 0)
    91	                # Ensure score is integer between 0 and 100
    92	                try:
    93	                    score = int(score)
    94	                    if score < 0:
    95	                        score = 0
    96	                    elif score > 100:
    97	                        score = 100
    98	                except (ValueError, TypeError):
    99	                    score = 0
   100	
   101	                alert_required = result.get("alert_required", False)
   102	                if not isinstance(alert_required, bool):
   103	                    alert_required = bool(alert_required)
   104	
   105	                reasoning = result.get("risk_reasoning", "No reasoning provided")
   106	                if not isinstance(reasoning, str):
   107	                    reasoning = str(reasoning)
   108	
   109	                # Update the finding dictionary with the risk scores
   110	                finding["policy_risk_score"] = score
   111	                finding["alert_required"] = alert_required
   112	                finding["risk_reasoning"] = reasoning
   113	                return finding
   114	            else:
   115	                print(f"[{self.agent_id}] No JSON object found in LLM response: {llm_response}")
   116	                return self._fallback_scoring(finding, text, entities)
   117	        except json_module.JSONDecodeError as e:
   118	            print(f"[{self.agent_id}] Failed to parse JSON from LLM response: {e}")
   119	            print(f"[{self.agent_id}] LLM response: {llm_response}")
   120	            return self._fallback_scoring(finding, text, entities)
   121	
   122	    def _fallback_scoring(self, finding, text, entities):
   123	        """
   124	        Fallback heuristic scoring if LLM fails.
   125	        Updates the finding dictionary in place and returns it.
   126	        """
   127	        text_lower = text.lower()
   128	        high_risk_keywords = [
   129	            "ban", "prohibit", "restrict", "investigation", "lawsuit", "fine",
   130	            "penalty", "sanction", "violation", "non-compliance", "enforcement",
   131	            "regulatory action", "cease and desist", "consent order", "settlement"
   132	        ]
   133	        keyword_count = sum(1 for keyword in high_risk_keywords if keyword in text_lower)
   134	        entity_count = len(entities.get("organizations", [])) + len(entities.get("policymakers", [])) + len(entities.get("themes", []))
   135	        score = min(100, (keyword_count * 10) + (entity_count * 5))
   136	        alert_required = score >= 80
   137	        reasoning = f"Fallback scoring: {keyword_count} high-risk keywords, {entity_count} entities."
   138	        
   139	        # Update the finding dictionary with the risk scores
   140	        finding["policy_risk_score"] = score
   141	        finding["alert_required"] = alert_required
   142	        finding["risk_reasoning"] = reasoning
   143	        return finding
   144	
   145	    def execute_task(self, payload):
   146	        findings = payload.get("findings", [])
   147	        scored_findings = []
   148	        
   149	        print(f"[{self.agent_id}] Scoring risk for {len(findings)} findings...")
   150	        
   151	        for finding in findings:
   152	            # Use compressed_summary if available, otherwise text_context
   153	            text_to_analyze = finding.get("compressed_summary", "")
   154	            if not text_to_analyze:
   155	                text_to_analyze = finding.get("text_context", "")
   156	            
   157	            entities = finding.get("entities", {"organizations": [], "policymakers": [], "themes": []})
   158	            
   159	            # Score the risk and update the finding in place
   160	            scored_finding = self._score_risk_with_llm(finding, text_to_analyze, entities)
   161	            
   162	            # Print critical alert if score >= 80
   163	            score = scored_finding.get("policy_risk_score", 0)
   164	            if score >= 80:
   165	                title = scored_finding.get("title", "Unknown Title")
   166	                print(f"[🚨 CRITICAL ALERT] High Risk Policy Detected: {title} - Score: {score}")
   167	            
   168	            scored_findings.append(scored_finding)
   169	        
   170	        return {"findings": scored_findings, "status": "COMPLETED"}

    11	    def _call_ollama(self, prompt):
    12	        payload = {
    13	            "model": self.model,
    14	            "prompt": prompt,
    15	            "stream": False
    16	        }
    17	        try:
    18	            response = requests.post(self.ollama_url, json=payload, timeout=10)
    19	            response.raise_for_status()
    20	            return response.json().get("response", "")
    21	        except Exception as e:
    22	            print(f"[{self.agent_id}] Error calling Ollama: {e}")
    23	            return None
    24	
    25	    def _parse_llm_response(self, response_text):
    26	        if not response_text:
    27	            return None
    28	            
    29	        # Strip whitespace
    30	        text = response_text.strip()
    31	        
    32	        # Try to extract JSON from markdown code blocks
    33	        # Pattern for ```json ... ``` or ``` ... ```
    34	        json_match = re.search(r'```(?:json)?\s*([\s\S]*?)\s*```', text, re.IGNORECASE)
    35	        if json_match:
    36	            json_str = json_match.group(1).strip()
    37	        else:
    38	            # Fallback: look for first { and last }
    39	            start = text.find('{')
    40	            end = text.rfind('}') + 1
    41	            if start >= 0 and end > start:
    42	                json_str = text[start:end]
    43	            else:
    44	                json_str = text
    45	        
    46	        # Try to parse as JSON
    47	        try:
    48	            return json.loads(json_str)
    49	        except json.JSONDecodeError:
    50	            # If JSON parsing fails, try to extract key-value pairs with regex
    51	            # Look for patterns like "validation_status": "approved" or 'validation_status': "approved"
    52	            result = {}
    53	            # Pattern for validation_status
    54	            status_match = re.search(r'"validation_status"\s*:\s*"(approved|rejected)"', json_str, re.IGNORECASE)
    55	            if status_match:
    56	                result["validation_status"] = status_match.group(1).lower()
    57	            # Pattern for false_positive_risk
    58	            risk_match = re.search(r'"false_positive_risk"\s*:\s*"(high|low)"', json_str, re.IGNORECASE)
    59	            if risk_match:
    60	                result["false_positive_risk"] = risk_match.group(1).lower()
    61	            # Pattern for confidence
    62	            conf_match = re.search(r'"confidence"\s*:\s*"(high|low)"', json_str, re.IGNORECASE)
    63	            if conf_match:
    64	                result["confidence"] = conf_match.group(1).lower()
    65	            
    66	            # If we found all three keys, return the result
    67	            if len(result) == 3:
    68	                return result
    69	            else:
    70	                print(f"[{self.agent_id}] Failed to parse LLM response: {response_text}")
    71	                return None

    78

    81	            impact_score = finding.get("impact_score", "low")

    90

    92	            title = finding.get("title", "")
    93	            context = finding.get("text_context", "")
    94	            impact_score = finding.get("impact_score", "")

    21	        """
    22	        Process and log human feedback on a policy record.
    23	
    24	        Expected payload:
    25	        {
    26	            "record_id": "unique identifier of the record",
    27	            "human_decision": one of ["accept", "reject", "monitor", "escalate", "false_positive", "needs_research"],
    28	            "reason": "explanation for the human decision",
    29	            "original_record": {...}  # optional, the original record being feedback on

    31	
    32	        Returns:
    33	            Dict with status and feedback file path.
    34	        """

    36	        human_decision = payload.get("human_decision")
    37	        reason = payload.get("reason", "")
    38	        original_record = payload.get("original_record", {})
    39	
    40	        # Validate human_decision
    41	        valid_decisions = ["accept", "reject", "monitor", "escalate", "false_positive", "needs_research"]
    42	        if human_decision not in valid_decisions:
    43	            return {
    44	                "agent_id": self.agent_id,
    45	                "status": "ERROR",
    46	                "message": f"Invalid human_decision: {human_decision}. Must be one of {valid_decisions}"
    47	            }
    48	
    49	        # Prepare feedback entry
    50	        feedback_entry = {
    51	            "timestamp": datetime.utcnow().isoformat() + "Z",
    52	            "record_id": record_id,
    53	            "human_decision": human_decision,
    54	            "reason": reason,
    55	            "original_record": original_record,
    56	            "agent_id": self.agent_id
    57	        }
    58	
    59	        # Append to JSONL file
    60	        try:
    61	            with open(self.feedback_file, 'a') as f:
    62	                f.write(json.dumps(feedback_entry) + '\n')
    63	            print(f"[{self.agent_id}] Feedback logged for record {record_id}: {human_decision}")
    64	            return {
    65	                "agent_id": self.agent_id,
    66	                "status": "SUCCESS",
    67	                "message": f"Feedback logged for record {record_id}",
    68	                "feedback_file": self.feedback_file
    69	            }
    70	        except Exception as e:
    71	            print(f"[{self.agent_id}] Error logging feedback: {e}")
    72	            return {

   196	                                   lookup_key=lookup_key,
   197	                                   swarm_id=self.SWARM_ID,
   198	                                   project_scope=self.PROJECT_SCOPE,
   199	                                   data=json.dumps(data),
   200	                                   updated_at=datetime.now(timezone.utc).isoformat(),
   201	                                   version=new_version)
   202	                record = result.single()
